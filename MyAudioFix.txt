# This is my way to force my rates and latency but keep in mind i dont look for low latency switches.
# My focus is to have less CPU usage for audio and have a smooth playback instead of dynamic latency.
# Pipwire-pulse tends to switch to 10ms or less when he wants.
# A reminder that default.clock.quantum=2048 is the max unless you set clock.power-of-two-quantum=false as it calculates it with a power of two (1024-2048-4096-8192) the minimum i believe is 0 auto but you might face a bunch of audio crackling bellow 1024...


/home/baz/.config/pipewire/pipewire.conf.d/pwrate.conf


context.properties = {
    default.clock.rate = 44100  # I care about less noise in the background (Dead Island 2 Seems to not like this rate...shame works perfectly under windows so ProtonGE/Cachy broke something unless Valve also broke it as well)
    default.clock.allowed-rates = [ 44100 48000 ]   # Allow this two only but 44.1khz is prefered
    default.clock.quantum = 3072  # 1024
    default.clock.min-quantum = 32  # 32
    default.clock.quantum-limit = 8192  # 8192
    default.clock.max-quantum = 8192  # 2048
    clock.power-of-two-quantum = false   # true (If you want a custom value set this to false)
    link.max-buffers = 64   # 64 / Could be a bigger value ? 3 clients can only support 16 buffers /More buffers is almost always worse than less, latency and memory wise.
    default.clock.quantum-floor = 8   # 4 / Minimum quantum to reserve space for. This is the minimum buffer size used in the graph, regardless of the samplerate.
    log.level = 0   # 2
    settings.check-quantum = false  # false
    settings.check-rate = fasle     # false
    mem.allow-mlock = true  # true
    mem.warn-mlock = false  # false
    mem.mlock-all = false   # false
    context.modules.allow-empty = false  # false / True if you want noise to be filtered out but better not since they might warn about an issue.
}

            #    block-sink-volume              blocks updates to sink volume
            #quirks = [ ]

             { application.process.binary = "teams" }
             { application.process.binary = "teams-insiders" }
             { application.process.binary = "teams-for-linux" }
             { application.process.binary = "skypeforlinux" }
        ]
        actions = { quirks = [ force-s16-info ] }
    }
    {
        # firefox marks the capture streams as don't move and then they
        # can't be moved with pavucontrol or other tools.
        matches = [ { application.process.binary = "firefox" } ]
        actions = { quirks = [ remove-capture-dont-move ] }
    }
    {
        # speech dispatcher asks for too small latency and then underruns.
        matches = [ { application.name = "~speech-dispatcher.*" } ]
        actions = {
            update-props = {
                pulse.min.req          = 2048/44100      # 10.6ms 512/48000
                pulse.min.quantum      = 2048/44100      # 10.6ms 512/48000
                pulse.idle.timeout     = 5              # pause after 5 seconds of underrun
            }
        }
    }

### api.alsa.period-num Batch / Non-Batch
# IRQ Frequency api.alsa.period-size/2 vs api.alsa.period-size
# Extra Delay api.alsa.headroom + api.alsa.period-size/2 vs api.alsa.headroom
# Buffer Size api.alsa.period-num * api.alsa.period-size/2 vs api.alsa.period-num * api.alsa.period-size

### Added a new fix for Wine/Proton audio fix even tho adds delay...as expected obviously but least the audio has less cracks in it plus less CPU wasted.

/home/baz/.config/pipewire/pipewire-pulse.conf.d/99-wine-quantum.conf

pulse.rules = [
  {
    matches = [
      { application.process.binary = "wine64-preloader" }
    ]
    actions = {
        quirks = [ force-s16-info ]
      update-props = {
        pulse.default.rate = 44100
        pulse.default.position = [ FL FR ]
        pulse.default.format = S16LE
        pulse.fix.format = "S16LE"
        pulse.default.req = 1024/44100
        pulse.min.req     = 1024/44100
        pulse.min.quantum = 1024/44100
        pulse.min.frag    = 1024/44100
      }
    }
  }
]
