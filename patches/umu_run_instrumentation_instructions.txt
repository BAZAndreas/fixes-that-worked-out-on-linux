A) monitor_windows (insert/replace the inner loops)

# Place inside monitor_windows, after the "Waiting for new windows IDs" debug and before the existing while not window_ids loop.
_mw_start = time.time()

while not window_ids:
    t0 = time.time()
    window_ids = get_pstree_window_ids(
        d_secondary, get_pstree_from_pid(pid), get_window_ids(d_secondary)
    )
    log.debug("monitor_windows: one-loop elapsed=%.3fs", time.time() - t0)

# ... later, in the main monitoring loop, replace the loop head with:
while True:
    t_loop = time.time()
    current_window_ids: set[int] | None = get_pstree_window_ids(
        d_secondary, get_pstree_from_pid(pid), get_window_ids(d_secondary, window_ids)
    )
    log.debug("monitor_windows: loop elapsed=%.3fs", time.time() - t_loop)
    if not current_window_ids:
        continue
    # (rest of function unchanged)

C) run_command (measure start/wait/total elapsed)

# Replace the existing Popen/proc.wait block with this block:
_t_start = time.time()
log.debug("run_command: starting command %s", command)
with Popen(command, start_new_session=True, cwd=cwd) as proc:  # nosec B603
    log.info("run_command: child started pid=%s", proc.pid)
    if is_steammode:
        ret = run_in_steammode(proc)
    else:
        t_wait_start = time.time()
        ret = proc.wait()
        log.debug("run_command: proc.wait time=%.3fs", time.time() - t_wait_start)
    log.debug("run_command: child %s exited with status=%s total_elapsed=%.3fs", proc.pid, ret, time.time() - _t_start)

return ret
